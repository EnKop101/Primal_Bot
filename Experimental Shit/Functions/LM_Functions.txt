from sentence_transformers import util
from sentence_transformers import SentenceTransformer
from transformers import pipeline

class LM_Functions:
    def __init__(self):
        self.zero_shot_classifier = pipeline("zero-shot-classification",model="cross-encoder/nli-deberta-v3-base")
        self.sentence_classifier = SentenceTransformer('sentence-transformers/multi-qa-mpnet-base-dot-v1')

    async def sentence_similarity(self, input, list_of_sentences):
        """
        This function calculates the similarity between two sentences using the cosine similarity method.
        """
        if self.zero_shot_classification(input):
            encoded_questions = self.sentence_classifier.encode(list_of_sentences)
            encoded_message = self.sentence_classifier.encode(input)

            scores = util.dot_score(encoded_message, encoded_questions)[0].cpu().tolist()
            max_score = max(scores)

            if max_score > 27.5:
                return list_of_sentences[scores.index(max_score)]
        else:
            return

    
    async def zero_shot_classification(self, input):
        """
        This function classifies a given input into one of the candidate labels.
        """
        candidate_labels = ["Question", "Request", "Demand"]
        result = self.zero_shot_classifier(input, candidate_labels)
        max_score = max(result['scores'])
        if max_score > 0.8:
            return True
        else:
            return False
