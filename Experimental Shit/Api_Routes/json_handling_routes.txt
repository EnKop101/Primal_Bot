from quart import jsonify, request
import Functions.Data_Handling

def register_json_handling_routes(app, dh: Functions.Data_Handling.Data_Handling, folder_path):
    @app.route("/qna/add-question", methods=["PUT"])
    async def add_question():
        data = await request.get_json()
        question = data.get("question", "")
        answer = data.get("answer", "")
        exact_sentence = data.get("exact_sentence", False)

        if question == "" or answer == "":
            return jsonify({"error": "Question or Answer not provided"}), 400

        if exact_sentence:
            await dh.Add_To_QnA_Json(question, answer, folder_path, True)
        else:
            await dh.Add_To_QnA_Json(question, answer, folder_path)
        
        return jsonify({"message": "Question added successfully"}), 200
    
    @app.route("/qna/delete-question", methods=["DELETE"])
    async def delete_question():
        data = await request.get_json()
        id = data.get("Id", 0)
        exact_sentence = data.get("exact_sentence", False)

        if id == 0:
            return jsonify({"error": "Question not provided"}), 400

        if exact_sentence:
            await dh.Delete_From_QnA_Json(id, folder_path, True)
        else:
            await dh.Delete_From_QnA_Json(id, folder_path)
        
        return jsonify({"message": "Question deleted successfully"}), 200
